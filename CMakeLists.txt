cmake_minimum_required(VERSION 3.22)
project(SwellPercussionist VERSION 0.4.1 LANGUAGES C CXX)
if(MSVC)
  set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
endif()
set(CMAKE_CXX_STANDARD 17)
include(FetchContent)
FetchContent_Declare(JUCE GIT_REPOSITORY https://github.com/juce-framework/JUCE.git GIT_TAG 8.0.6 GIT_SHALLOW TRUE)
FetchContent_MakeAvailable(JUCE)
# Monocypher (BSD-2 / CC0) : vérification Ed25519 des fichiers de licence.
# Pas de CMakeLists chez Monocypher : SOURCE_SUBDIR inexistant = télécharger seulement, la cible est déclarée ici.
FetchContent_Declare(monocypher GIT_REPOSITORY https://github.com/LoupVaillant/Monocypher.git GIT_TAG 4.0.3 GIT_SHALLOW TRUE SOURCE_SUBDIR none)
FetchContent_MakeAvailable(monocypher)
add_library(monocypher STATIC "${monocypher_SOURCE_DIR}/src/monocypher.c" "${monocypher_SOURCE_DIR}/src/optional/monocypher-ed25519.c")
target_include_directories(monocypher PUBLIC "${monocypher_SOURCE_DIR}/src" "${monocypher_SOURCE_DIR}/src/optional")
set_target_properties(monocypher PROPERTIES POSITION_INDEPENDENT_CODE ON)
set(PERC_FORMATS VST3 Standalone)
if(APPLE)
  list(APPEND PERC_FORMATS AU)
endif()
# Never change PLUGIN_CODE / BUNDLE_ID once released: saved DAW projects would lose the plugin.
juce_add_plugin(SwellPercussionist COMPANY_NAME "Swell-audio" BUNDLE_ID "fr.swell-audio.percussionist" PLUGIN_MANUFACTURER_CODE Swel PLUGIN_CODE Sprc FORMATS ${PERC_FORMATS} PRODUCT_NAME "Swell The Percussionist" ICON_BIG "${CMAKE_CURRENT_SOURCE_DIR}/Resources/icon.png" IS_SYNTH TRUE NEEDS_MIDI_INPUT TRUE NEEDS_MIDI_OUTPUT FALSE COPY_PLUGIN_AFTER_BUILD FALSE)
# Photos of the tabs, and the sounds built by Tools/pack_samples.py.
file(GLOB PERC_PHOTOS CONFIGURE_DEPENDS assets/*.jpg)
file(GLOB PERC_SAMPLES CONFIGURE_DEPENDS assets/samples/pack/*.flac)
juce_add_binary_data(PercAssets SOURCES ${PERC_PHOTOS} ${PERC_SAMPLES})
target_sources(SwellPercussionist PRIVATE Source/PluginProcessor.cpp Source/PluginEditor.cpp Source/Licence/SwellLicence.cpp)
target_compile_definitions(SwellPercussionist PUBLIC SWELL_VERSION="${PROJECT_VERSION}" JUCE_WEB_BROWSER=0 JUCE_USE_CURL=0 JUCE_VST3_CAN_REPLACE_VST2=0)
target_compile_options(SwellPercussionist PUBLIC $<$<CXX_COMPILER_ID:MSVC>:/utf-8>)
target_link_libraries(SwellPercussionist PRIVATE PercAssets monocypher juce::juce_audio_utils juce::juce_dsp juce::juce_cryptography PUBLIC juce::juce_recommended_config_flags)
enable_testing()
juce_add_console_app(PercTest PRODUCT_NAME "Percussionist Audio Tests")
target_sources(PercTest PRIVATE Tests/AudioTest.cpp)
target_include_directories(PercTest PRIVATE "${juce_SOURCE_DIR}/modules")
target_compile_definitions(PercTest PRIVATE JUCE_GLOBAL_MODULE_SETTINGS_INCLUDED=1)
target_link_libraries(PercTest PRIVATE SwellPercussionist)
add_test(NAME PercAudio COMMAND PercTest)
juce_add_console_app(PercSnapshot PRODUCT_NAME "Percussionist Snapshot")
target_sources(PercSnapshot PRIVATE Tools/Snapshot.cpp)
target_include_directories(PercSnapshot PRIVATE "${juce_SOURCE_DIR}/modules")
target_compile_definitions(PercSnapshot PRIVATE JUCE_GLOBAL_MODULE_SETTINGS_INCLUDED=1)
target_link_libraries(PercSnapshot PRIVATE SwellPercussionist)
